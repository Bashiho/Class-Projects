CREATE TABLE Textbook(
	title TEXT,
	ISBN BIGINT,
	pageCount INTEGER,
	price REAL,
	copyright TEXT,
	edition INTEGER,
	printDate DATE,
	publisherID INTEGER
);

CREATE TABLE Publisher(
	publisherID INTEGER,
	name TEXT,
	email TEXT
);

CREATE TABLE Author(
	authorID INTEGER,
	firstName TEXT,
	lastName TEXT,
	email TEXT
);

CREATE TABLE AuthorXBook(
	ISBN BIGINT,
	authorID INTEGER
);
